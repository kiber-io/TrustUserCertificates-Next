cmake_minimum_required(VERSION 3.28)
project(tuc_hooks LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LSPLANT_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(LSPLANT_STANDALONE OFF CACHE BOOL "" FORCE)
set(DOBBY_DEBUG OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory(../vendor/lsplant/lsplant/src/main/jni lsplant EXCLUDE_FROM_ALL)
add_subdirectory(../vendor/dobby dobby EXCLUDE_FROM_ALL)
set(XZ_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../vendor/xz-embedded)
add_library(tuc-xz STATIC ${XZ_ROOT}/linux/lib/xz/xz_crc32.c ${XZ_ROOT}/linux/lib/xz/xz_crc64.c
    ${XZ_ROOT}/linux/lib/xz/xz_dec_stream.c ${XZ_ROOT}/linux/lib/xz/xz_dec_lzma2.c)
target_include_directories(tuc-xz PUBLIC ${XZ_ROOT}/linux/include/linux PRIVATE ${XZ_ROOT}/userspace)
target_compile_definitions(tuc-xz PUBLIC XZ_USE_CRC64)
add_library(tuc-hooks SHARED hooks.cpp)
target_include_directories(tuc-hooks PRIVATE ../vendor/lsplant/lsplant/src/main/jni/include ../vendor/dobby/include)
target_compile_options(tuc-hooks PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
target_link_libraries(tuc-hooks PRIVATE lsplant_static dobby_static tuc-xz log dl)
target_link_options(tuc-hooks PRIVATE -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384)
